## 本机环境

人守急停。不要写 `curl ...\; echo`（zsh 会请求 `/health;`）。

SMT 车：

```Bash
NAV=http://192.168.71.51:8001
ZM=http://192.168.71.51:8081
```

中免 Helios 当前也在 `.51`（和 SMT 不要同时开）。若走 130 网再改成 `.105`。

```Bash
wait_ok() {
  local rid="$1" i st
  for i in $(seq 1 90); do
    sleep 1
    st="$(curl -sS "$NAV/status/$rid" | python3 -c 'import json,sys; d=json.load(sys.stdin); print(d.get("terminal_state") or d.get("state") or "")')"
    echo "  $rid  $st"
    case "$st" in
      SUCCEEDED) return 0 ;;
      FAILED|REJECTED|CANCELLED|TIMED_OUT) echo "失败 $rid"; return 1 ;;
    esac
  done
  echo "超时 $rid"
  return 1
}
goto_wait() {
  local st="$1"
  local rid="CHK-${st}-$(date +%H%M%S)-$RANDOM"
  echo "=== $st $rid ==="
  curl -sS -X POST "$NAV/goto" -H 'Content-Type: application/json' \
    -d "{\"task_id\":\"CHECK\",\"request_id\":\"$rid\",\"timeout_sec\":90,\"idempotency_key\":\"$rid\",\"station_id\":\"$st\"}"
  echo
  wait_ok "$rid"
}
nav_to() {
  local id="$1"
  local key="${id}-$(date +%H%M%S)-$RANDOM"
  echo "=== $id  key=$key ==="
  curl -sS -w "\nHTTP %{http_code}\n" -X POST "$ZM/navigation/navigate" \
    -H 'Content-Type: application/json' \
    -H "Idempotency-Key: $key" \
    -d "{\"nav_id\":\"$id\"}"
}
same_key() {
  local id="$1"
  local key="${id}-$(date +%H%M%S)-$RANDOM"
  echo "=== $id  key=$key 第一次 ==="
  curl -sS -w "\nHTTP %{http_code}\n" -X POST "$ZM/navigation/navigate" \
    -H 'Content-Type: application/json' \
    -H "Idempotency-Key: $key" \
    -d "{\"nav_id\":\"$id\"}"
  echo
  echo "=== $id  key=$key 第二次（应马上 200、车不动） ==="
  curl -sS -w "\nHTTP %{http_code}\n" -X POST "$ZM/navigation/navigate" \
    -H 'Content-Type: application/json' \
    -H "Idempotency-Key: $key" \
    -d "{\"nav_id\":\"$id\"}"
}
```

`/goto` 刚返回往往是 `RUNNING`，必须等到 `SUCCEEDED` 才算到站。停车：

```Bash
curl -sS -X POST "$NAV/stop" -H 'Content-Type: application/json' -d '{}'
```

空闲时厂家可能 `330001`，可忽略。
